  coinbase balance                                # Test CLI
  python3 verify_coinbase_v3.py                  # Full test suite

USAGE - SIMPLE BALANCE CHECK
  python3 check_balance.py                        # Table format
  python3 check_balance.py --json                 # JSON format
  python3 check_balance.py --csv                  # CSV format

USAGE - PYTHON CODE
  from trading_system.connectors.coinbase_v3 import CoinbaseConnectorV3
  
  cb = CoinbaseConnectorV3()
  
  # Get balances
  balances = cb.get_balances()
  
  # Get price
  price = cb.get_price('BTC-USD')
  
  # Preview order (no execution)
  preview = cb.preview_order(
      product_id='BTC-USD',
      side='BUY',
      order_type='market',
      quote_size=100.0
  )
  
  # Execute order
  order = cb.create_order(
      product_id='BTC-USD',
      side='BUY',
      order_type='market',
      quote_size=100.0
  )

TROUBLESHOOTING
  coinbase: command not found
    → npm install -g @coinbase/coinbase-cli
  
  HTTP 401 - Unauthorized
    → Create new ECDSA key in CDP Portal
    → Run setup script with new key
  
  HTTP 403 - Missing scopes
    → Add Trade + Transfer permissions to key
    → Create new key with correct permissions
  
  No module named 'trading_system'
    → Run from portfolio-management/ directory
    → python3 check_balance.py (not from subdirectory)

COMMON COMMANDS
  coinbase balance                  # Check account balance
  coinbase products get BTC-USD     # Get BTC price
  coinbase orders list              # List all orders
  coinbase balance --jq '.USD'      # Get just USD balance
  coinbase --help                   # CLI help
